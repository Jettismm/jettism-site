// ---- Jettism VOD Library backend ---- // Deploy this as a Web App (Execute as: Me, Who has access: Anyone). // Paste the resulting /exec URL into vods.html as APPS_SCRIPT_URL. const SHEET_NAME = "VODs"; // tab name in your Google Sheet const ACCESS_PASSWORD = "ProductionTeam"; // shared password for both site login and adding VODs const MAX_ATTEMPTS = 3; const BAN_MINUTES = 5; // GET requests must include ?password=...&deviceId=... — the list itself // is gated, not just the page, so the data can't be pulled directly by URL. function doGet(e) { const password = (e.parameter && e.parameter.password) || ""; const deviceId = (e.parameter && e.parameter.deviceId) || "unknown"; const check = checkPassword(deviceId, password); if (!check.ok) return check.response; return jsonResponse({ success: true, vods: getAllVods() }); } function doPost(e) { const body = JSON.parse(e.postData.contents); const deviceId = body.deviceId || "unknown"; const check = checkPassword(deviceId, body.password || ""); if (!check.ok) return check.response; if (body.action === "login") { // just confirming the password is correct — no data changes return jsonResponse({ success: true }); } if (body.action === "update_timestamps") { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME); const data = sheet.getDataRange().getValues(); let rowIndex = -1; for (let i = 1; i < data.length; i++) { if (normalizeDate(data[i][2]) === body.date) { rowIndex = i + 1; // sheet rows are 1-indexed, +1 more for header already skipped by loop start break; } } if (rowIndex === -1) { return jsonResponse({ success: false, error: "Couldn't find a stream with that date." }); } sheet.getRange(rowIndex, 6).setValue(JSON.stringify(body.timestamps || [])); // column F = timestamps return jsonResponse({ success: true, vods: getAllVods() }); } // default / "add" action const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME); const entry = body.entry || {}; sheet.appendRow([ entry.title || "", "", // channel column kept for backward compatibility with existing rows, no longer used entry.date || "", entry.duration || "TBD", entry.download || "", JSON.stringify(entry.timestamps || []) ]); return jsonResponse({ success: true, vods: getAllVods() }); } // Shared password check + rate limiting for both login and add-VOD flows. // Returns { ok: true } on success, or { ok: false, response: } on failure. function checkPassword(deviceId, password) { const cache = CacheService.getScriptCache(); const banKey = "vod_ban_" + deviceId; const failKey = "vod_fail_" + deviceId; if (cache.get(banKey)) { return { ok: false, response: jsonResponse({ success: false, banned: true, error: "Too many attempts. Try again in a few minutes." }) }; } if (password !== ACCESS_PASSWORD) { const fails = Number(cache.get(failKey) || 0) + 1; if (fails >= MAX_ATTEMPTS) { cache.put(banKey, "1", BAN_MINUTES * 60); cache.remove(failKey); return { ok: false, response: jsonResponse({ success: false, banned: true, error: "Too many attempts. Try again in a few minutes." }) }; } cache.put(failKey, String(fails), 300); return { ok: false, response: jsonResponse({ success: false, error: "Incorrect password." }) }; } cache.remove(failKey); return { ok: true }; } function getAllVods() { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME); const data = sheet.getDataRange().getValues(); const rows = data.slice(1); // skip header row return rows .filter(r => r[0]) // skip blank rows .map(r => ({ title: r[0], date: normalizeDate(r[2]), duration: normalizeDuration(r[3]), download: r[4], timestamps: safeParse(r[5]) })); } // Sheets silently converts date-looking text into real Date objects on // entry (paste or type), even in columns with no explicit date format. // This converts whatever comes back — a Date object or a plain string — // into a consistent "YYYY-MM-DD" string the page expects. function normalizeDate(value) { if (Object.prototype.toString.call(value) === '[object Date]') { return Utilities.formatDate(value, Session.getScriptTimeZone(), "yyyy-MM-dd"); } return String(value || "").trim(); } // Same issue for durations like "2:58:25" — Sheets treats them as a // time-of-day value. Converts that back into "H:MM:SS" text, or passes // through plain strings ("TBD", already-correct text) unchanged. function normalizeDuration(value) { if (Object.prototype.toString.call(value) === '[object Date]') { const h = value.getHours(); const m = value.getMinutes(); const s = value.getSeconds(); return h > 0 ? `${h}:${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}` : `${m}:${String(s).padStart(2, '0')}`; } return String(value || "TBD").trim(); } function safeParse(value) { try { return JSON.parse(value || "[]"); } catch (err) { return []; } } function jsonResponse(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
